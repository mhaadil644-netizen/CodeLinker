        // Add to cart
        function addToCart(productId, size = 10) {
            const product = products.find(p => p.id === productId);
            const price = size === 10 ? product.price : 
                        size === 20 ? Math.round(product.price * 1.8) :
                        size === 50 ? Math.round(product.price * 4) :
                        Math.round(product.price * 7);
            
            const existingItem = cart.find(item => item.productId === productId && item.size === size);
            
            if (existingItem) {
                existingItem.quantity += 1;
            } else {
                cart.push({
                    productId,
                    name: product.name,
                    size,
                    price,
                    quantity: 1
                });
            }
            
            updateCartCount();
            showNotification(`${product.name} (${size}ml) added to cart!`);
        }

        // Add to cart from modal
        function addToCartFromModal(productId) {
            addToCart(productId, selectedSize);
        }

        // Update cart count
        function updateCartCount() {
            const count = cart.reduce((total, item) => total + item.quantity, 0);
            document.getElementById('cartCount').textContent = count;
        }

        // Toggle cart modal
        function toggleCart() {
            const modal = document.getElementById('cartModal');
            modal.classList.toggle('hidden');
            
            if (!modal.classList.contains('hidden')) {
                renderCartItems();
            }
        }

        // Render cart items
        function renderCartItems() {
            const container = document.getElementById('cartItems');
            const total = cart.reduce((sum, item) => sum + (item.price * item.quantity), 0);
            
            if (cart.length === 0) {
                container.innerHTML = '<p class="text-gray-500 text-center py-8">Your cart is empty</p>';
            } else {
                container.innerHTML = cart.map(item => `
                    <div class="flex items-center justify-between p-4 border rounded-lg">
                        <div class="flex-1">
                            <h4 class="font-semibold">${item.name}</h4>
                            <p class="text-sm text-gray-600">${item.size}ml - $${item.price}</p>
                        </div>
                        <div class="flex items-center space-x-3">
                            <button onclick="updateQuantity(${item.productId}, ${item.size}, -1)" class="w-8 h-8 bg-gray-200 rounded-full flex items-center justify-center hover:bg-gray-300">-</button>
                            <span class="font-semibold">${item.quantity}</span>
                            <button onclick="updateQuantity(${item.productId}, ${item.size}, 1)" class="w-8 h-8 bg-gray-200 rounded-full flex items-center justify-center hover:bg-gray-300">+</button>
                            <button onclick="removeFromCart(${item.productId}, ${item.size})" class="text-red-500 hover:text-red-700 ml-2">🗑️</button>
                        </div>
                    </div>
                `).join('');
            }
            
            document.getElementById('cartTotal').textContent = `$${total.toFixed(2)}`;
        }

        // Update quantity
        function updateQuantity(productId, size, change) {
            const item = cart.find(item => item.productId === productId && item.size === size);
            if (item) {
                item.quantity += change;
                if (item.quantity <= 0) {
                    removeFromCart(productId, size);
                } else {
                    updateCartCount();
                    renderCartItems();
                }
            }
        }

        // Remove from cart
        function removeFromCart(productId, size) {
            cart = cart.filter(item => !(item.productId === productId && item.size === size));
            updateCartCount();
            renderCartItems();
        }

        // Checkout
        function checkout(paymentMethod) {
            if (cart.length === 0) {
                showNotification('Your cart is empty!', 'error');
                return;
            }
            
            const total = cart.reduce((sum, item) => sum + (item.price * item.quantity), 0);
            const orderNumber = 'ORD' + Date.now().toString().slice(-6);
            
            if (paymentMethod === 'online') {
                showNotification(`Redirecting to secure payment... Order #${orderNumber} - Total: $${total.toFixed(2)}`);
                // Simulate payment processing
                setTimeout(() => {
                    cart = [];
                    updateCartCount();
                    toggleCart();
                    showNotification('Payment successful! Order confirmed.', 'success');
                }, 2000);
            } else {
                showNotification(`Order placed! Order #${orderNumber} - Total: $${total.toFixed(2)} - Cash on Delivery`, 'success');
                cart = [];
                updateCartCount();
                toggleCart();
            }
        }
